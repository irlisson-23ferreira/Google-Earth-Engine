// DEFININDO AS COORDENADAS DE FORTALEZA
var lon = -38.5434;
var lat = -3.71839;
var ponto = ee.Geometry.Point([lon, lat]); 

// APROXIMANDO PARA FORTALEZA
Map.centerObject(ponto, 10);

// DEFININDO AS IMAGENS A SEREM UTILIZADAS
var colecao = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
  .filterBounds(ponto) // FILTRO DE ÁREA
  .filterDate('2023-01-01','2023-06-30') // FILTRO DE DATA
  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE',10)); // FILTRO DE COBERTURA DE NUVENS

// APLICANDO A MEDIANA NOS PIXELS DAS IMAGENS ENCONTRADAS
var imagemMediana = colecao.median();

// APLICANDO O NDVI NA IMAGEM RESULTADO DA MEDIANA
var ndvi = imagemMediana.normalizedDifference(['B8', 'B4']).rename('NDVI');

// DEFININDO OS PARÂMETROS DE VISUALIZAÇÃO DA IMAGEM DO NDVI
var visParamsNDVI = {
  min: 0,
  max: 0.8,
  palette: ['red','yellow','green']
};

// VISUALIZANDO A IMAGEM DE NDVI
Map.addLayer(ndvi, visParamsNDVI, 'Indice de Vegetação (NDVI)');


// EXPORTANDO A IMAGEM
// 1. CRIANDO A REGIÃO DE CORTE
// Opção A: Um raio de 15 km ao redor do ponto (ajuste o valor se necessário)
var regionArea = ponto.buffer(15000); 

// Opção B (alternativa): Usar os limites do próprio mapa visível no momento
// var regionArea = Map.getBounds(true);

// 2. EXPORTANDO COM O PARÂMETRO 'region'
Export.image.toDrive({
  image: ndvi,
  description: 'vic_image',
  folder: 'earthengine',
  fileNamePrefix: 'vic_image',
  scale: 10,
  region: regionArea, // <--- ADICIONADO AQUI
  maxPixels: 1e11
});
